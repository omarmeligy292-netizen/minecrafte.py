# ألوان البلوكات
BLOCKS = {
    "Grass": color.rgb(90, 180, 65),
    "Dirt": color.rgb(130, 85, 50),
    "Stone": color.rgb(130, 130, 140),
    "Wood": color.rgb(150, 100, 50),
    "Sand": color.rgb(220, 200, 130),
}

selected_block = "Grass"
world = {}

# إنشاء مكعب
class Voxel(Button):
    def _init_(self, position=(0, 0, 0), block_type="Grass"):
        super()._init_(
            parent=scene,
            position=position,
            model="cube",
            origin_y=0.5,
            texture="white_cube",
            color=BLOCKS[block_type],
            highlight_color=color.lime,
            collider="box"
        )

        self.block_type = block_type
        self.position_key = tuple(int(v) for v in position)
        world[self.position_key] = self

    def input(self, key):
        global selected_block

        if not self.hovered:
            return

        if key == "left mouse down":
            # كسر البلوك
            if self in world.values():
                world.pop(self.position_key, None)
                destroy(self)

        elif key == "right mouse down":
            # وضع بلوك بجانب البلوك الذي تشير إليه
            new_position = self.position + mouse.normal
            new_position = tuple(
                int(round(v)) for v in new_position
            )

            if new_position not in world:
                # لا تسمح بوضع بلوك داخل اللاعب
                player_pos = player.position
                distance = Vec3(*new_position).distance_to(player_pos)

                if distance > 1.5:
                    Voxel(
                        position=new_position,
                        block_type=selected_block
                    )


# إنشاء العالم
random.seed(7)

WORLD_SIZE = 16

for x in range(-WORLD_SIZE // 2, WORLD_SIZE // 2):
    for z in range(-WORLD_SIZE // 2, WORLD_SIZE // 2):
        # ارتفاع بسيط للأرض
        height = random.choice([0, 0, 0, 1])

        for y in range(-2, height + 1):
            if y == height:
                block = "Grass"
            elif y >= height - 2:
                block = "Dirt"
            else:
                block = "Stone"

            Voxel(position=(x, y, z), block_type=block)
